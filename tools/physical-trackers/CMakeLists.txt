cmake_minimum_required(VERSION 3.21)
project(vdxr_physical_trackers LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This prototype uses Windows local shared memory")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(tracker-ipc-test ipc_test.cpp)
enable_testing()
add_test(NAME tracker-ipc COMMAND tracker-ipc-test)
set(VDXR_RUNTIME_DLL "" CACHE FILEPATH "Built x64 VDXR DLL for synthetic integration test")
if(VDXR_RUNTIME_DLL)
  add_executable(tracker-quest-probe quest_probe.cpp)
  target_link_libraries(tracker-quest-probe PRIVATE d3d11 dxgi)
  target_include_directories(tracker-quest-probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../external/OpenXR-SDK/include")
  add_executable(tracker-runtime-test runtime_test.cpp)
  target_include_directories(tracker-runtime-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../external/OpenXR-SDK/include")
  add_test(NAME tracker-runtime COMMAND tracker-runtime-test "${VDXR_RUNTIME_DLL}")
  set_tests_properties(tracker-ipc tracker-runtime PROPERTIES RUN_SERIAL TRUE)
endif()

set(SURVIVE_SOURCE "" CACHE PATH "Pinned libsurvive source checkout")
set(SURVIVE_LIBRARY "" CACHE FILEPATH "Matching double-precision survive import library")
if(SURVIVE_SOURCE AND SURVIVE_LIBRARY)
  add_executable(vdxr-survive provider.cpp)
  target_include_directories(vdxr-survive PRIVATE "${SURVIVE_SOURCE}/include"
    "${SURVIVE_SOURCE}/redist" "${SURVIVE_SOURCE}/libs/cnmatrix/include")
  target_compile_definitions(vdxr-survive PRIVATE USE_DOUBLE SURVIVE_EXPORT=__declspec\(dllimport\))
  target_link_libraries(vdxr-survive PRIVATE "${SURVIVE_LIBRARY}")
  if(TARGET tracker-quest-probe)
    target_include_directories(tracker-quest-probe PRIVATE "${SURVIVE_SOURCE}/include"
      "${SURVIVE_SOURCE}/redist" "${SURVIVE_SOURCE}/libs/cnmatrix/include")
    target_compile_definitions(tracker-quest-probe PRIVATE VDXR_WITH_SURVIVE USE_DOUBLE SURVIVE_EXPORT=__declspec\(dllimport\))
    target_link_libraries(tracker-quest-probe PRIVATE "${SURVIVE_LIBRARY}")
  endif()
endif()
